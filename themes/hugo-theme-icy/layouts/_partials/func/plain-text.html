{{- /* Readable plain text of a page's content, for the search index. Unlike .Plain it drops theme UI that render hooks and shortcodes add inside the content (heading anchors, copy buttons, player controls, tab bars…): those are wrapped in … comment markers by the templates that emit them. Footnote back-links are removed too, and entities are decoded. Params: page. Returns string. */ -}} {{- $html := .Content | string }} {{- $html = replaceRE `(?s).*?` " " $html }} {{- $html = replaceRE `(?s)]*class="footnote-backref"[^>]*>.*?` "" $html }} {{- $html = replaceRE `` "$0 " $html }} {{- return ($html | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace) }}
