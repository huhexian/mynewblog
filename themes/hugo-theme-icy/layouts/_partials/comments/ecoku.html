{{- /* Ecoku comments. The markup follows the contract of Ecoku's client/ecoku-loader.js (data-ecoku-* hooks); comments.js injects that loader only when the section approaches the viewport, so readers who never scroll to the comments make no request to the comment server. The loader handles timeouts, error messages and retry itself; colors map to the theme's tokens in components/comments-ecoku.css. Params (dict): page, cfg (params.comments.ecoku): serverURL (required) e.g. https://comments.example.com siteId (required) scriptURL loader URL (default /client/ecoku-loader.js) cssURL Ecoku stylesheet. Default "none": Ecoku loads no CSS and the theme's components/comments-ecoku.css styles everything. Set a URL (e.g. /client/ecoku.css) to use Ecoku's own. pageSize comments per page (1–100, default 10) theme auto | light | dark (default auto: follows the page) i18n en | zh-CN | zh-Hant (default: from the page's language) */ -}} {{- $page := .page }} {{- $cfg := .cfg }} {{- if not (and $cfg.serverURL $cfg.siteId) }} {{- errorf "comments: ecoku needs params.comments.ecoku.serverURL and siteId" }} {{- end }} {{- $server := strings.TrimSuffix "/" $cfg.serverURL }} {{- $script := $cfg.scriptURL | default (printf "%s/client/ecoku-loader.js" $server) }} {{- /* Ecoku's interface language follows the page: en, zh-Hant or zh-CN. Other languages use the Ecoku site's own default. */}} {{- $i18n := $cfg.i18n | default "" }} {{- if and $i18n (not (in (slice "en" "zh-CN" "zh-Hant") $i18n)) }} {{- errorf "comments: params.comments.ecoku.i18n must be en, zh-CN or zh-Hant, got %q" $i18n }} {{- end }} {{- if not $i18n }} {{- $locale := lower ($page.Language.Locale | default $page.Language.Lang) }} {{- if hasPrefix $locale "en" }}{{ $i18n = "en" }} {{- else if or (hasPrefix $locale "zh-hant") (in (slice "zh-tw" "zh-hk" "zh-mo") $locale) }}{{ $i18n = "zh-Hant" }} {{- else if hasPrefix $locale "zh" }}{{ $i18n = "zh-CN" }} {{- end }} {{- end }} {{- partial "assets/use.html" (dict "page" $page "feature" "comments") }}
